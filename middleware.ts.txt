import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'

export function middleware(request: NextRequest) {
  const { searchParams } = request.nextUrl
  
  // 1. Проверяем, есть ли в URL параметр открытия модалки
  const isClosingPost = searchParams.get('close-post') === 'true'

  if (isClosingPost) {
    // 2. Извлекаем данные пользователя (например, из куки)
    // В реальном проекте тут может быть проверка JWT или сессии
    const userRole = request.cookies.get('user-role')?.value

    // 3. Если роли нет или она не подходит
    if (userRole !== 'TUBE_DASH' && userRole !== 'ADMIN') {
      const url = request.nextUrl.clone()
      
      // Удаляем "опасные" параметры из URL
      url.searchParams.delete('close-post')
      url.searchParams.delete('summary-id')
      url.searchParams.delete('post-id')
      url.searchParams.delete('post-title')

      // Редиректим на ту же страницу, но с чистым URL
      return NextResponse.redirect(url)
    }
  }

  return NextResponse.next()
}

// Настраиваем, на каких путях должен работать этот фильтр
export const config = {
  matcher: '/dash/:path*', 
}
